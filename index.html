El código

Cómo funciona el código

El programa tiene 3 bloques principales — cada uno tiene un propósito claro:

① Variables y pines — van ANTES del setup(), al inicio del programa
#include "Arduino_SensorKit.h"  // librería del kit Seeed

// ── ¿En qué pin está cada componente? ──────────────
const int pinLuz    = A3;  // sensor de luz → puerto A3 Grove
const int pinSuelo  = A1;  // sensor de suelo → pin lateral A1
const int pinBuzzer = 5;   // buzzer → puerto D5 Grove
const int pinLED    = 6;   // LED → puerto D6 Grove
// El OLED va por I2C — no necesita declarar pin

// ── Umbrales de decisión ────────────────────────────
const int NOCHE = 30;   // luz < 30 = oscuro = planta duerme
const int SECO  = 750;  // suelo < 750 = húmedo = feliz

// ── Rango real para calcular % en pantalla ──────────
const int SUELO_HUMEDO = 530;   // valor con tierra regada
const int SUELO_SECO   = 1023;  // valor con tierra seca
⚠️ Regla clave: si cambias el sensor de pin físico en el protoboard, DEBES cambiar también el número en el código. Si el código dice A3 y el cable está en A1 — no funciona nada.
② setup() — configuración inicial, corre UNA sola vez al encender el Arduino
void setup() {

  // Pines que ENVÍAN señal → OUTPUT
  pinMode(pinBuzzer, OUTPUT); // buzzer envía
  pinMode(pinLED,    OUTPUT); // LED envía

  // Los sensores son INPUT (leen)
  // analogRead() ya lo sabe solo

  // Iniciar Monitor Serie
  Serial.begin(9600);

  // Iniciar pantalla OLED
  Oled.begin();
  Oled.setFlipMode(true);
  Oled.setFont(u8x8_font_chroma48medium8_r);

  // Mensaje de bienvenida
  Oled.clear();
  Oled.setCursor(1, 3);
  Oled.print("Hola! Soy tu planta");
  delay(2000); // espera 2 segundos
}

③ loop() — se repite PARA SIEMPRE mientras haya energía
void loop() {
  // 1. Leer los sensores
  int luz   = analogRead(pinLuz);    // 0 = oscuro, 770 = sol
  int suelo = analogRead(pinSuelo);  // 530 = húmedo, 1023 = seco

  noTone(pinBuzzer); // apagar buzzer al inicio de cada ciclo

  // 2. Decidir qué estado mostrar
  if (luz < NOCHE) {                     // DUERME si está oscuro
    carita("-_- zZ", "Durmiendo...", suelo);
    digitalWrite(pinLED, HIGH);          // enciende LED como luna

  } else if (suelo < SECO) {             // FELIZ si tierra húmeda
    carita("^_^", "Estoy feliz!", suelo);
    digitalWrite(pinLED, LOW);

  } else {                               // SED si tierra seca
    carita("T_T", "Tengo sed!", suelo);
    tone(pinBuzzer, 440, 150);           // pita una vez
    digitalWrite(pinLED, LOW);
  }

  delay(1000); // esperar 1 segundo antes de repetir
}

// Función que dibuja en el OLED
void carita(const char* ojos, const char* estado, int suelo) {
  int pct = map(suelo, SUELO_SECO, SUELO_HUMEDO, 0, 100);
  pct = constrain(pct, 0, 100);
  Oled.clear();
  Oled.setCursor(3, 0);  Oled.print(ojos);    // renglón 0: carita
  Oled.setCursor(0, 3);  Oled.print(estado);  // renglón 3: estado
  Oled.setCursor(0, 6);
  Oled.print("Hum: "); Oled.print(pct); Oled.print("%");
}
